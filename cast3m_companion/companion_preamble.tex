% Exercises numbered C<chapter>.<n>, so that they never collide with the
% exercises of the book they refer to (Exercise 5.12 vs Exercise C5.2).
\renewcommand{\theexo}{C\thechapter.\arabic{exo}}

% A gibiane operator or keyword in the running text (as in the Cast3M
% introduction).
\providecommand{\op}[1]{\texttt{#1}}

%---- shell listings (gibiane listings: style gibstyle, in preamble/style) --
\lstdefinestyle{shstyle}{language=bash,
  basicstyle=\ttfamily\scriptsize,
  keywordstyle=\color{blue!60!black},commentstyle=\color{gray},
  backgroundcolor=\color{gray!6},frame=none,
  breaklines=true,showstringspaces=false,
  xleftmargin=0.6em,aboveskip=4pt,belowskip=4pt}

%---- worked examples: explanation on the left, gibiane on the right ------
% Copied from the Cast3M introduction (preamble/style.tex, "Cast3M-specific"),
% so that the two documents show gibiane the same way. Keep castemcom and
% castemlines adjacent, with NO blank line between them. Lines of at most
% about 50 characters fit the 92 mm box.
\colorlet{CodeGray}{gray!12}
\newlength{\castemcomwidth}   \setlength{\castemcomwidth}{65.mm}
\newlength{\castemlineswidth} \setlength{\castemlineswidth}{92.mm}
\newlength{\codeboxsep}       \setlength{\codeboxsep}{5.pt}
\newsavebox{\castemsavebox}
\newenvironment{castemlines}
{\begin{lrbox}{\castemsavebox}%
 \begin{minipage}[t]{\dimexpr\castemlineswidth-2\codeboxsep\relax}%
 \begin{small}}
{\end{small}%
 \end{minipage}%
 \end{lrbox}%
 {\setlength{\fboxsep}{\codeboxsep}\colorbox{CodeGray}{\usebox{\castemsavebox}}}%
 \par\medskip\noindent}
\newenvironment{castemcom}
{\par\vspace*{3.mm}%
 \noindent
 \begin{minipage}[t]{\castemcomwidth}%
 \begin{small}}
{\end{small}%
 \end{minipage}%
 \hspace*{4.mm}}
